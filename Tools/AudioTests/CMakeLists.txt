cmake_minimum_required(VERSION 3.24)
project(EsteriaAudioOutputTest CXX C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The audio fixture exercises the real 64-bit Windows FMOD backend")
endif()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${REPO}/cmake/TargetArchitecture.cmake")
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(WIN32 _WINDOWS WOWDECOMP_ARCH_64=1 _CRT_SECURE_NO_WARNINGS)
if(MSVC AND WOWDECOMP_CPU_ARM64)
    add_compile_options(/volatile:ms)
endif()
add_subdirectory("${REPO}/Engine/Source/SoundEngine" "fmod-source")
add_executable(fmod_output_device fmod_output_device.cpp)
target_link_libraries(fmod_output_device PRIVATE fmodex user32)
