cmake_minimum_required(VERSION 3.20)
project(SkyridingTests CXX)
enable_testing()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(skyriding_tests skyriding_tests.cpp)
target_compile_features(skyriding_tests PRIVATE cxx_std_17)
target_include_directories(skyriding_tests PRIVATE "${REPO}")
add_test(NAME skyriding_tests COMMAND skyriding_tests)
set_tests_properties(skyriding_tests PROPERTIES TIMEOUT 30)
