cmake_minimum_required(VERSION 3.24)
project(EsteriaPlatformTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(CLIENT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_test(NAME target_architecture COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_SOURCE_DIR}/test_architecture.cmake")
add_executable(test_lznt1 test_lznt1.cpp "${CLIENT_ROOT}/WoW/Common/ComponentCore/Lznt1.cpp")
target_include_directories(test_lznt1 PRIVATE "${CLIENT_ROOT}")
add_test(NAME lznt1_catalog COMMAND test_lznt1)
foreach (backend native scalar)
    add_executable(test_float4_${backend} test_float4.cpp)
    target_include_directories(test_float4_${backend} PRIVATE "${CLIENT_ROOT}")
    if (backend STREQUAL "scalar")
        target_compile_definitions(test_float4_${backend} PRIVATE WOWDECOMP_SIMD_SCALAR=1)
    endif ()
    if (NOT MSVC)
        target_compile_options(test_float4_${backend} PRIVATE -ffp-contract=off)
    endif ()
    add_test(NAME float4_${backend} COMMAND test_float4_${backend})
endforeach ()
option(WOWDECOMP_TEST_XVID "Build/test the portable movie codec from pinned source" ON)
if (WOWDECOMP_TEST_XVID)
    include("${CLIENT_ROOT}/cmake/XvidPortable.cmake")
    add_executable(test_xvid test_xvid.cpp)
    target_link_libraries(test_xvid PRIVATE XvidPortable)
    add_test(NAME xvid_movie_roundtrip COMMAND test_xvid)
endif ()
