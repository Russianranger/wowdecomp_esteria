cmake_minimum_required(VERSION 3.20)
project(OsMemoryTests CXX)
enable_testing()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(os_alloc_near os_alloc_near.cpp "${REPO}/Engine/Source/Os/Win64/OsMemory.cpp")
target_compile_features(os_alloc_near PRIVATE cxx_std_17)
target_include_directories(os_alloc_near PRIVATE "${REPO}")
target_compile_definitions(os_alloc_near PRIVATE WOWDECOMP_ARCH_64=1 WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME os_alloc_near COMMAND os_alloc_near)
set_tests_properties(os_alloc_near PROPERTIES TIMEOUT 15)
