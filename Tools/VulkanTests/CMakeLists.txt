cmake_minimum_required(VERSION 3.20)
project(VulkanTextureTests CXX)
enable_testing()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(texture_registry texture_registry.cpp
    "${REPO}/Engine/Source/Gx/CGxDeviceVulkan/Vulkan/Texture/Registry.cpp")
target_compile_features(texture_registry PRIVATE cxx_std_20)
target_include_directories(texture_registry PRIVATE "${REPO}" "${REPO}/ThirdParty/vulkan-headers")
target_compile_definitions(texture_registry PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME texture_registry COMMAND texture_registry)

option(VULKAN_TEXTURE_DEVICE_TEST "Exercise image growth on the local Vulkan device" OFF)
option(VULKAN_PRESENT_DEVICE_TEST "Clear and present through the real Win32 Vulkan surface" OFF)
if(VULKAN_TEXTURE_DEVICE_TEST OR VULKAN_PRESENT_DEVICE_TEST)
    if(NOT WIN32)
        message(FATAL_ERROR "Device fixtures exercise the Windows Vulkan backend; build Windows binaries")
    endif()
    file(GLOB_RECURSE TIER "${REPO}/Engine/Source/Gx/CGxDeviceVulkan/Vulkan/*.cpp")
    list(FILTER TIER EXCLUDE REGEX "/Log.cpp$")
    add_library(vulkan_fixture_backend OBJECT ${TIER})
    target_compile_features(vulkan_fixture_backend PUBLIC cxx_std_20)
    target_include_directories(vulkan_fixture_backend PUBLIC "${REPO}" "${REPO}/ThirdParty/vulkan-headers")
    target_compile_definitions(vulkan_fixture_backend PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(vulkan_fixture_backend PUBLIC user32)
endif()
if(VULKAN_TEXTURE_DEVICE_TEST)
    add_executable(texture_device texture_device.cpp)
    target_link_libraries(texture_device PRIVATE vulkan_fixture_backend)
    add_test(NAME texture_device COMMAND texture_device)
    set_tests_properties(texture_device PROPERTIES TIMEOUT 180)
endif()
if(VULKAN_PRESENT_DEVICE_TEST)
    add_executable(presentation_device presentation_device.cpp)
    target_link_libraries(presentation_device PRIVATE vulkan_fixture_backend)
    add_test(NAME presentation_device COMMAND presentation_device)
    set_tests_properties(presentation_device PROPERTIES TIMEOUT 120)
endif()
